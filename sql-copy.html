-- =====================================================================
-- Diamond Shop — CUSTOM diamond orders (1,000 to 100,000 diamonds)
-- Run this ONCE in the Supabase SQL Editor (Sir runs SQL himself).
--
-- Pricing (computed SERVER-SIDE so it can never be faked by the site):
--   Blended average of the 4 public packages:
--     (90+160+250+400) taka / (90+180+270+449) diamonds = 900/989 = 0.91/diamond
--   Volume tiers (bigger order = cheaper per diamond):
--     1,000 - 9,999     -> 0.91 per diamond (average rate)
--     10,000 - 49,999   -> 0.86 per diamond (~5.5% discount)
--     50,000 - 100,000  -> 0.82 per diamond (~10% discount)
--   amount = round(diamonds * rate)
--
-- Orders are inserted with pkg = 'custom' so the existing admin page
-- and complete_order() crediting work exactly as before.
-- =====================================================================

create or replace function public.place_custom_order(
  p_username text, p_diamonds integer, p_trxid text, p_phone text
)
returns json language plpgsql security definer set search_path = public as $$
declare
  u text := norm_username(p_username);
  t text := upper(trim(p_trxid));
  ph text := regexp_replace(coalesce(p_phone, ''), '[\s-]', '', 'g');
  amt integer;
  oid uuid;
begin
  if u is null or length(u) < 2 or length(u) > 24 then
    raise exception 'invalid username';
  end if;
  if p_diamonds is null or p_diamonds < 1000 or p_diamonds > 100000 then
    raise exception 'custom amount must be 1000-100000';
  end if;
  if t is null or length(t) < 6 or length(t) > 24 or t !~ '^[A-Z0-9]+$' then
    raise exception 'invalid trxid';
  end if;
  if ph !~ '^(\+8801[3-9][0-9]{8}|01[3-9][0-9]{8})$' then
    raise exception 'invalid phone';
  end if;

  amt := round(p_diamonds * (case
    when p_diamonds >= 50000 then 0.82
    when p_diamonds >= 10000 then 0.86
    else 0.91 end));

  -- create the profile if this username is brand new (+30 welcome bonus)
  insert into public.profiles (username, diamonds, owned)
  values (u, 30, '{}')
  on conflict (username) do nothing;

  insert into public.orders (username, pkg, diamonds, amount, trxid, phone, status)
  values (u, 'custom', p_diamonds, amt, t, ph, 'pending')
  returning id into oid;
  return json_build_object('ok', true, 'orderId', oid, 'amount', amt);
exception when unique_violation then
  raise exception 'duplicate trxid';
end;
$$;

grant execute on function public.place_custom_order(text, integer, text, text) to anon, authenticated;
